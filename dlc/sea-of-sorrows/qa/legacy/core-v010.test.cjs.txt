function progressTo(g,index){for(let i=0;i<index;i++){const o=O[objectiveOrder[i]];g.state.flags[o.flag]=true;g.state.completed.push(o.id);}g.state.objectiveIndex=index;g.state.inventory={pistol:true,foam:true,flame:true,tracker:true};}
function route(g,to){const old=g.state.room,q=[[old]],seen=new Set();while(q.length){const path=q.shift(),id=path.at(-1);if(id===to){g.state.room=old;return path;}if(seen.has(id))continue;seen.add(id);g.state.room=id;for(const d of R[id].doors)if(g.doorStatus(d).open)q.push([...path,d.to]);}g.state.room=old;return null;}
function go(g,to){const path=route(g,to);assert.ok(path,`No legal route ${g.state.room}→${to} at ${g.state.objectiveIndex}`);for(const id of path.slice(1)){const door=g.room.doors.find(d=>d.to===id&&g.doorStatus(d).open);assert.ok(door);g.p.x=door.x;g.p.z=g.zFor(door);g.cooldowns.interaction=0;const nearby=g.nearby();assert.equal(nearby.action,'door',`Door obscured by ${nearby.action} in ${g.state.room}`);g.interact();assert.equal(g.state.room,id);}}
function shoot(g,p,target,weapon='pistol'){g.p.x=p.x;g.p.z=g.mode==='SIDE'?.55:p.z;g.p.weapon=weapon;g.state.inventory[weapon]=true;g.cooldowns.fire=0;return g.shoot(target);}
function freshRoom(id,mode='HYBRID'){const g=new Game({presentation:mode});g.enter(id);g.drainEvents();return g;}
